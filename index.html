using System; namespace AdventureGame { class Program { static void Main(string[] args) { Console.OutputEncoding = System.Text.Encoding.UTF8; ShowWelcomeScreen(); int x = 0; int y = 0; int gems = 0; bool isPlaying = true; int mapSize = 6; bool gemExists = true; int gemX = 2; int gemY = 2; Console.Clear(); while (isPlaying) { // الإضافة هنا: تنظيف الشاشة في بداية كل دور لضمان عدم تكرار رسم الخريطة Console.Clear(); Console.WriteLine("\n--- The Adventurous Bird Game ---"); Console.WriteLine("Goal: Reach the point (5,5)"); for (int i = mapSize; i >= 0; i--) { for (int j = 0; j <= mapSize; j++) { if (x == j && y == i) Console.Write(" 🐥 "); else if (j == 5 && i == 5) Console.Write(" 🚩 "); else if (gemExists && j == gemX && i == gemY) Console.Write(" 💎 "); else if (j == 1 && i == 3) Console.Write(" 👾 "); else Console.Write(" . "); } Console.WriteLine(); } Console.WriteLine($"Gems: {gems} | Position: {x},{y}"); Console.WriteLine("Move (w/a/s/d):"); string move = Console.ReadLine(); switch (move) { case "w": y++; break; case "s": y--; break; case "d": x++; break; case "a": x--; break; default: Console.WriteLine("Invalid move!"); break; } if (gemExists && x == gemX && y == gemY) { Console.WriteLine("Great! You collected a 💎!"); gems++; gemExists = false; } if (x == 1 && y == 3) { ShowGameOverScreen(); isPlaying = false; } if (x == 5 && y == 5) { ShowWinScreen(gems); isPlaying = false; } } Console.WriteLine("Game Closed"); } static void ShowWelcomeScreen() { Console.Clear(); Console.WriteLine("╔════════════════════════════════════════════╗"); Console.WriteLine("║ ║"); Console.WriteLine("║ Welcome to the game of the adventurous bird ║"); Console.WriteLine("║ ║"); Console.WriteLine("╚════════════════════════════════════════════╝"); Console.WriteLine("\nPress ENTER to start..."); Console.ReadLine(); } static void ShowGameOverScreen() { Console.Clear(); Console.WriteLine("╔══════════════════════════════╗"); Console.WriteLine("║ GAME OVER ║"); Console.WriteLine("╠══════════════════════════════╣"); Console.WriteLine("║ 👾 The monster caught you! ║"); Console.WriteLine("║ Better luck next time... ║"); Console.WriteLine("╚══════════════════════════════╝"); Console.WriteLine("\nPress ENTER to exit..."); Console.ReadLine(); } static void ShowWinScreen(int gems) { Console.Clear(); Console.WriteLine("╔══════════════════════════════╗"); Console.WriteLine("║ YOU WIN! 🎉 ║"); Console.WriteLine("╠══════════════════════════════╣"); Console.WriteLine("║ You reached the 🚩 flag! ║"); Console.WriteLine($"║ Total Gems: {gems} ║"); Console.WriteLine("║ Amazing job! ║"); Console.WriteLine("╚══════════════════════════════╝"); Console.WriteLine("\nPress ENTER to exit..."); Console.ReadLine(); } } }
